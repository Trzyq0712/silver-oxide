# A divide-by-zero guard whose divisor is a struct field reached through a `&mut`.
#
#     pub fn body_apply_impulse(b: &mut Body, j: &Vec2) {
#         if 0 < b.mass { b.vel.x = b.vel.x + j.x / b.mass; ... }
#     }
#
# The guard's read of `b.mass` and the obligation's read land in different e-classes

# not to the write between the divisions, and not to there being two of them:
#
#     divisor a parameter          (`param_div`, `param_div_pure`)   verifies
